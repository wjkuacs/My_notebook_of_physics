\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        width=10cm, height=8cm,
        xlabel={$T$},
        ylabel={$p$},
        xmin=0, xmax=5,
        ymin=0, ymax=5,
        axis lines=middle,
        xtick=\empty,
        ytick=\empty,
        every axis plot/.append style={very thick}
    ]
    % 汽化曲线
    \addplot[blue,domain=0.5:4.5,samples=100] {0.1*exp(2.5*\x)};
    \node[blue,above] at (axis cs:4,4.5) {汽化曲线};
    
    % 升华曲线
    \addplot[red,domain=0.2:2,samples=100] {0.05*exp(3*\x)};
    \node[red,left] at (axis cs:1.5,2.5) {升华曲线};
    
    % 熔化曲线
    \addplot[green!60!black,domain=1:3,samples=100] {0.3 + 0.5*\x};
    \node[green!60!black,right] at (axis cs:2.5,1.6) {熔化曲线};
    
    % 三相点
    \fill (axis cs:1.5,1.05) circle (3pt);
    \node[right] at (axis cs:1.5,1.05) {三相点};
    
    % 临界点
    \fill (axis cs:3.5,3.2) circle (3pt);
    \node[above] at (axis cs:3.5,3.2) {临界点};
    
    % 区域标注
    \node at (axis cs:1,3.5) {气态};
    \node at (axis cs:3,0.5) {液态};
    \node at (axis cs:1,0.3) {固态};
    
    % 箭头标注
    \draw[->,thick] (axis cs:2,3) -- (axis cs:2.5,2.5);
    \node[above] at (axis cs:2.25,2.75) {临界温度};
    \end{axis}
\end{tikzpicture}
\end{document}
