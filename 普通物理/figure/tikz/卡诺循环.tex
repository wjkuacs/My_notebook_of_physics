\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        width=10cm, height=8cm,
        xlabel={$V$},
        ylabel={$p$},
        xmin=0, xmax=5,
        ymin=0, ymax=5,
        axis lines=middle,
        xtick=\empty,
        ytick=\empty,
        every axis plot/.append style={very thick}
    ]
    % 等温线 T1 > T2
    \addplot[blue,domain=0.5:5,samples=100] {4/x};
    \node[blue,right] at (axis cs:4.5,0.9) {$T_1$};
    
    \addplot[red,dashed,domain=0.8:5,samples=100] {2.5/x};
    \node[red,right] at (axis cs:4.5,0.55) {$T_2$};
    
    % 卡诺循环 A->B->C->D->A
    \coordinate (A) at (axis cs:1,4);
    \coordinate (B) at (axis cs:2,2);
    \coordinate (C) at (axis cs:3,0.83);
    \coordinate (D) at (axis cs:1.5,1.67);
    
    % 等温膨胀 A->B
    \draw[very thick,green!60!black,->] (A) -- (B) node[midway,above left]{$等温膨胀$};
    % 绝热膨胀 B->C
    \draw[very thick,orange,->] (B) -- (C) node[midway,right]{$绝热膨胀$};
    % 等温压缩 C->D
    \draw[very thick,green!60!black,->] (C) -- (D) node[midway,below right]{$等温压缩$};
    % 绝热压缩 D->A
    \draw[very thick,orange,->] (D) -- (A) node[midway,left]{$绝热压缩$};
    
    % 标注点
    \fill (A) circle (2pt) node[above left]{$A$};
    \fill (B) circle (2pt) node[right]{$B$};
    \fill (C) circle (2pt) node[right]{$C$};
    \fill (D) circle (2pt) node[left]{$D$};
    
    % 热源标注
    \node[below] at (axis cs:2.5,0) {$T_1 > T_2$};
    \end{axis}
\end{tikzpicture}
\end{document}
