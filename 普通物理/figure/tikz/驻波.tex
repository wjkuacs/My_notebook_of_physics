\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        width=14cm, height=6cm,
        xlabel={$x$},
        ylabel={$\psi$},
        xmin=0, xmax=6.5,
        ymin=-2.2, ymax=2.2,
        axis lines=middle,
        xtick=\empty,
        ytick=\empty,
        every axis plot/.append style={thick}
    ]
    % 驻波 t=0 时刻
    \addplot[blue,domain=0:6.5,samples=300] {2*sin(deg(pi*x))*cos(deg(0))};
    
    % 驻波 t=T/8 时刻
    \addplot[red,dashed,domain=0:6.5,samples=300] {2*sin(deg(pi*x))*cos(deg(45))};
    
    % 包络线
    \addplot[gray,dotted,domain=0:6.5,samples=200] {2*sin(deg(pi*x))};
    \addplot[gray,dotted,domain=0:6.5,samples=200] {-2*sin(deg(pi*x))};
    
    % 波节标注
    \foreach \x in {0,1,2,3,4,5,6} {
        \fill (axis cs:\x,0) circle (2pt);
        \draw[dashed,gray,thin] (axis cs:\x,-2.2) -- (axis cs:\x,2.2);
    }
    \node[below] at (axis cs:0,-2.2) {$0$};
    \node[below] at (axis cs:1,-2.2) {$\frac{\lambda}{2}$};
    \node[below] at (axis cs:2,-2.2) {$\lambda$};
    \node[below] at (axis cs:3,-2.2) {$\frac{3\lambda}{2}$};
    \node[below] at (axis cs:4,-2.2) {$2\lambda$};
    
    % 波节和波腹标注
    \node[above] at (axis cs:0.5,2.2) {波腹};
    \node[below] at (axis cs:1,-2.2) {波节};
    \node[above] at (axis cs:1.5,2.2) {波腹};
    
    % 图例
    \node[right,blue] at (axis cs:5.5,1.8) {$t=0$};
    \node[right,red] at (axis cs:5.5,1.3) {$t=T/8$};
    \end{axis}
\end{tikzpicture}
\end{document}
