\documentclass[border=5pt]{standalone}
\usepackage{tikz}
\usepackage{amsmath}
\usepackage{pgfplots}
\pgfplotsset{compat=1.18}
\begin{document}
\begin{tikzpicture}
    \begin{axis}[
        width=12cm, height=6cm,
        xlabel={$x$},
        ylabel={$\psi$},
        xmin=-1, xmax=5,
        ymin=-0.6, ymax=1.2,
        axis lines=middle,
        xtick=\empty,
        ytick=\empty,
        every axis plot/.append style={thick}
    ]
    % 势垒
    \fill[gray!20] (axis cs:1,0) rectangle (axis cs:2,1.2);
    \node[above] at (axis cs:1.5,1.2) {$V_0$};
    \draw[very thick] (axis cs:1,0) -- (axis cs:1,1.2);
    \draw[very thick] (axis cs:2,0) -- (axis cs:2,1.2);
    
    % 入射波
    \addplot[blue,domain=-1:1,samples=200] {0.8*sin(deg(5*pi*x)) + 0.3*cos(deg(5*pi*x))};
    \node[blue,above] at (axis cs:-0.5,0.8) {入射波};
    
    % 势垒内衰减波
    \addplot[red,domain=1:2,samples=100] {0.3*exp(-2*(\x-1))};
    \node[red,above] at (axis cs:1.5,0.4) {衰减波};
    
    % 透射波
    \addplot[green!60!black,domain=2:5,samples=200] {0.15*sin(deg(5*pi*x)) + 0.06*cos(deg(5*pi*x))};
    \node[green!60!black,above] at (axis cs:4,0.15) {透射波};
    
    % 反射波
    \addplot[blue,dashed,domain=-1:0.8,samples=100] {-0.2*sin(deg(5*pi*x)) + 0.1*cos(deg(5*pi*x))};
    \node[blue,above] at (axis cs:-0.8,-0.2) {反射波};
    
    % 区域标注
    \node[below] at (axis cs:0,0) {I};
    \node[below] at (axis cs:1.5,0) {II};
    \node[below] at (axis cs:3.5,0) {III};
    \end{axis}
\end{tikzpicture}
\end{document}
